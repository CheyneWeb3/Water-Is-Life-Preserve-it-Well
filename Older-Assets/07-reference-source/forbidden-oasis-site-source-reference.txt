(function(){
  const fmt=n=>Math.round(n).toLocaleString('en-US');

  const ABI=[
    "function totalSupply() view returns (uint256)",
    "function decimals() view returns (uint8)"
  ];

  function countUp(el,target,dur){
    const start=performance.now();
    function step(now){
      const p=Math.min((now-start)/dur,1);
      const e=1-Math.pow(1-p,3);
      el.textContent=fmt(target*e);
      if(p<1)requestAnimationFrame(step);else el.textContent=fmt(target);
    }
    requestAnimationFrame(step);
  }

  function render(original,evaporated){
    const remaining=Math.max(original-evaporated,0);
    countUp(origEl,original,1600);
    setTimeout(()=>countUp(evapEl,evaporated,2200),300);
    setTimeout(()=>countUp(remainEl,remaining,2200),300);
    const pct=original>0?(evaporated/original*100):0;

      fillEl.style.width=Math.min(pct,100)+'%';
      pctEl.textContent=pct.toFixed(2)+'% EVAPORATED';

  // This token uses a DEFLATIONARY True Burn: _burn() destroys tokens and lowers totalSupply().
  // So:  remaining water = totalSupply()  |  evaporated = original supply - totalSupply()
  async function fetchOnChain(){
    const addr=(CONTRACT_ADDRESS||"").trim();
    if(!addr || typeof ethers==='undefined') return null;
    try{
      const provider=new ethers.providers.JsonRpcProvider(BSC_RPC);
      const token=new ethers.Contract(addr,ABI,provider);
      let decimals=18;
      try{ decimals=await token.decimals(); }catch(e){ /* default 18 */ }

      // Current circulating supply straight from the contract (this is "remaining water").
      const supplyBN=await token.totalSupply();
      const toNum=bn=>parseFloat(ethers.utils.formatUnits(bn,decimals));
      const circulating=toNum(supplyBN);

      // The original minted supply is fixed and not recoverable on-chain after burns,
      // so it comes from the config baseline.
      const original=TOTAL_SUPPLY_FALLBACK;

      // Evaporated = everything destroyed since launch.
      const evaporated=Math.max(original-circulating,0);

      // Sanity guard: if the RPC returned a bogus/zero supply, bail to fallback.
      if(circulating<=0 || circulating>original*1.0001) return null;
      return {original, evaporated};
    }catch(err){
      console.warn('[Oasis] On-chain fetch failed, using fallback:',err);
      return null;
    }
  }

  // Quiet refresh: short tween only if a value actually changed, no big count-up.
  let lastEvap=null;
  function quietUpdate(original,evaporated){
    if(evaporated===lastEvap) return;       // nothing burned since last poll
    const fromEvap=(lastEvap==null)?evaporated:lastEvap;
    lastEvap=evaporated;
    const remaining=Math.max(original-evaporated,0);
    const fromRemain=Math.max(original-fromEvap,0);
    tween(evapEl,fromEvap,evaporated,900);

    fillEl.style.width=Math.min(pct,100)+'%';

  function tween(el,from,to,dur){
    const start=performance.now();
    function step(now){
      const p=Math.min((now-start)/dur,1);
      const e=1-Math.pow(1-p,3);
      el.textContent=fmt(from+(to-from)*e);
      if(p<1)requestAnimationFrame(step);else el.textContent=fmt(to);
    }
    requestAnimationFrame(step);
  }

  let fired=false, pollTimer=null;
  const sanct=document.getElementById('sanctuary');

  function poll(){
    fetchOnChain().then(data=>{ if(data) quietUpdate(data.original,data.evaporated); });
  }
  function startPolling(){
    if(pollTimer || !(CONTRACT_ADDRESS||"").trim()) return;
    pollTimer=setInterval(()=>{ if(!document.hidden) poll(); }, BURN_REFRESH_MS);
  }
  // Pause polling while tab is hidden; refresh once on return.
  document.addEventListener('visibilitychange',()=>{ if(!document.hidden && fired) poll(); });

  const evapIO=new IntersectionObserver(es=>{
    es.forEach(en=>{
      if(en.isIntersecting&&!fired){
        fired=true;
        evapIO.disconnect();
        // Show fallback immediately for instant drama, then refine with live data.
        render(TOTAL_SUPPLY_FALLBACK, EVAPORATED_FALLBACK);
        lastEvap=EVAPORATED_FALLBACK;
        fetchOnChain().then(data=>{
          if(data){ render(data.original, data.evaporated); lastEvap=data.evaporated; }
          startPolling();   // begin live updates after first load
        });
      }
    });
  },{threshold:0.35});
  evapIO.observe(sanct);
})();

/* ============ CONTRACT ADDRESS — COPY (revealed at launch) ============ */
(function(){
  const wrap=document.getElementById('contractAddr');
  const txt=document.getElementById('caText');
  const btn=document.getElementById('caCopy');
  const addr=(CONTRACT_ADDRESS||"").trim();

  // Launch gate: the Forge address stays PENDING until the countdown hits zero,
  // even if the address is entered early. (Everything else using the address stays live.)
  const launchTs=new Date(typeof LAUNCH_UTC!=='undefined'?LAUNCH_UTC:'2026-07-28T14:00:00Z').getTime();
  const launched=()=> isNaN(launchTs) ? true : (Date.now()>=launchTs);

  function showPending(){
    txt.textContent='PENDING';
    wrap.dataset.pending="true";
    if(btn) btn.hidden=true;
  }

  function reveal(){
    if(!addr){ showPending(); return; }
    txt.textContent=addr;
    wrap.dataset.pending="false";
    if(btn){
      btn.hidden=false;
      btn.onclick=async()=>{
        try{
          if(navigator.clipboard&&navigator.clipboard.writeText){
            await navigator.clipboard.writeText(addr);
          }else{
            const t=document.createElement('textarea');t.value=addr;t.style.position='fixed';t.style.opacity='0';
            document.body.appendChild(t);t.select();document.execCommand('copy');document.body.removeChild(t);
          }
          const lbl=btn.querySelector('.ca-copy-label');const orig=lbl.textContent;
          btn.classList.add('copied');lbl.textContent='Copied';
          setTimeout(()=>{btn.classList.remove('copied');lbl.textContent=orig;},1800);
        }catch(e){
          const lbl=btn.querySelector('.ca-copy-label');lbl.textContent='Copy failed';
          setTimeout(()=>{lbl.textContent='Copy';},1800);
        }
      };
    }
  }

  if(addr && launched()){
    reveal();
  }else if(addr && !launched()){
    // address entered early — keep it hidden until the timer runs out, then reveal live.
    showPending();
    const wait=launchTs-Date.now();
    setTimeout(reveal, Math.max(wait,0));
  }else{
    showPending();
  }

  // Swap button stays live whenever an address is set (independent of the launch gate).
  const swap=document.getElementById('swapCta');
  if(swap){
    if(addr){
      swap.href='https://pancakeswap.finance/swap?outputCurrency='+addr;
    }else{
      swap.href='#';
      swap.addEventListener('click',e=>{
        e.preventDefault();
        swap.classList.add('flash');setTimeout(()=>swap.classList.remove('flash'),4500);
      });
    }
  }
})();
